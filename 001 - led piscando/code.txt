int ledPin = 3; // GPIO 4 
void setup() { 
pinMode(ledPin, OUTPUT); 
} 
void loop() { 
digitalWrite(ledPin, HIGH); 
delay(500); 
digitalWrite(ledPin, LOW); 
delay(500); 

digitalWrite(ledPin, HIGH); 
delay(500); 
digitalWrite(ledPin, LOW); 
delay(500); 

digitalWrite(ledPin, HIGH); 
delay(500); 
digitalWrite(ledPin, LOW); 
delay(500); 

delay(2000);
}
