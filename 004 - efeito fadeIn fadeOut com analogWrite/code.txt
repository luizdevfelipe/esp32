int led1 = 2;
int led2 = 1;
int led3 = 3;

int i = 0;
bool fadeIn = true;
void setup() {
  pinMode(led1, OUTPUT);
  pinMode(led2, OUTPUT);
  pinMode(led3, OUTPUT);
}
void loop() {

  if (i < 255 && fadeIn) {
    i++;
  } else {
    fadeIn = false;
    if (i > 0) {
      i --;
    } else {
      fadeIn = true;
    }
  }

  analogWrite(led1, i);
  analogWrite(led2, i);
  analogWrite(led3, i);
  delay(10);
}
